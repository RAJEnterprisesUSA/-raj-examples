const POP60={baby:4.4, lil:4.0, unc:4.2, vaeh:4.6, head:4.8};
function lockT60(g){ return g.tw1+0.1+0.35; }
function tableT60(){ ctx.save(); ctx.translate(540,FLOOR); ctx.scale(SC60,SC60); ctx.translate(-540,-FLOOR); drawTable(540,0,0); ctx.restore(); }
function chairT60(cx,face,k){
  if(k<=0) return; const kk=easeOutBack(k);
  ctx.save(); ctx.translate(cx,FLOOR); ctx.scale(SC60*kk*(face===1?-1:1),SC60*kk); ctx.translate(-cx,-FLOOR); drawChair(cx,0,0,0); ctx.restore();
}
function handT60(px,face,sk,Yh,lean,A,back){
  const sx=back?-6:30, sy=back?-202:-198;
  const [ex,ey]=seg(sx,sy,A.sh,100), [hx,hy]=seg(ex,ey,A.sh+A.el,96);
  const th=rad(-lean), c=Math.cos(th), s=Math.sin(th);
  return [px+face*(hx*c-hy*s)*sk, Yh+(hx*s+hy*c)*sk];
}
// pose of one guest at time t: walking in, hopping onto the chair, then seated
function poseT60(t,g){
  if(t<g.tw0) return null;
  const standY=FLOOR-238*g.sk, seatY=FLOOR-SEATHIP60*SC60, xs=g.cx+g.face*SEATX60*SC60;
  const step=Math.sin(t*11);
  if(t<g.tw1+0.1){
    const u=ease((t-g.tw0)/(g.tw1-g.tw0));
    return {x:lerp(g.x0,xs,u), Yh:standY-Math.abs(step)*14*g.sk, k:0, lean:0,
      legs:(t<g.tw1)?[{hip:20*step,knee:26+16*Math.max(0,-step)},{hip:-20*step,knee:26+16*Math.max(0,step)}]:[{hip:6,knee:4},{hip:-6,knee:4}],
      arm:{sh:24+14*step,el:30}};
  }
  const k=ease((t-g.tw1-0.1)/0.35);
  return {x:xs, Yh:lerp(standY,seatY,k)-Math.sin(Math.PI*k)*40*g.sk, k:k, lean:lerp(0,-14,k),
    legs:[{hip:lerp(6,84,k),knee:lerp(4,-86,k)},{hip:lerp(-6,78,k),knee:lerp(-4,-80,k)}],
    arm:{sh:lerp(24,92,k),el:lerp(30,-4,k)}};
}
function drawGuestT60(t,g,P,extra){
  const p=Object.assign({x:P.x, y:FLOOR-(FLOOR-P.Yh)/g.sk, face:g.face, legs:P.legs, arm:P.arm, lean:P.lean, mouth:g.mouth, style:g.style}, extra||{});
  kidWithProp(p, g.sk);
}
function pillsT60(t){
  const t0=9.0; if(t<t0) return;
  const kk=easeOutBack(Math.min(1,(t-t0)/0.4)), ids=['lil','baby','unc','vaeh'], w=225, gap=15, x0=(1080-(4*w+3*gap))/2, y=448, h=78;
  ctx.save(); ctx.translate(540,y+h/2); ctx.scale(kk,kk); ctx.translate(-540,-(y+h/2));
  ids.forEach((id,i)=>{
    const g=G60[id], lt=lockT60(g), locked=(t>=lt), x=x0+i*(w+gap);
    const pk=locked?(1+0.12*Math.max(0,1-(t-lt)/0.3)):1;
    ctx.save(); ctx.translate(x+w/2,y+h/2); ctx.scale(pk,pk); ctx.translate(-(x+w/2),-(y+h/2));
    ctx.fillStyle=locked?'rgba(24,96,58,.94)':'rgba(26,15,43,.92)'; rr(x,y,w,h,20); ctx.fill();
    ctx.strokeStyle=locked?'#2fbf71':GOLD; ctx.lineWidth=4; rr(x,y,w,h,20); ctx.stroke();
    ctx.textAlign='center'; ctx.lineJoin='round';
    ctx.font='600 32px Oswald'; ctx.lineWidth=6; ctx.strokeStyle=INK; ctx.strokeText(g.n,x+w/2,y+36); ctx.fillStyle='#fff'; ctx.fillText(g.n,x+w/2,y+36);
    ctx.font='600 24px Oswald';
    const lab=locked?'LOCKED IN':'PICKING'+'...'.slice(0,1+Math.floor(t*3)%3);
    ctx.fillStyle=locked?'#aef3c9':GOLD; ctx.fillText(lab,x+w/2,y+66);
    ctx.restore();
  });
  ctx.restore();
}

}
