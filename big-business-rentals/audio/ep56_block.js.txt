const BUB56=[ // launch time, label, source, guests added

const SRC56={vaeh:[988,1190,'#7fd3ff'], lil:[100,1090,'#b8f0a0'], baby:[880,1200,'#ffc2e0'], cat:[988,1190,'#ffffff']};
function guestsAt56(tt){ let n=36, last=-9; for(const b of BUB56){ if(tt>=b[0]+FLY56){ n+=b[3]; last=b[0]+FLY56; } } return [n,last]; }
function tablesOf56(n){ return Math.ceil(n/6); }
function tableTime56(i){ if(i<6) return 0.3+i*0.15; let n=36; for(const b of BUB56){ n+=b[3]; if(tablesOf56(n)>i) return b[0]+FLY56; } return 99; }
function lastTableChange56(tt){ let n=36, last=-9; for(const b of BUB56){ const before=tablesOf56(n); n+=b[3]; if(tt>=b[0]+FLY56 && tablesOf56(n)>before) last=b[0]+FLY56; } return last; }
function numK56(txt,cx,base,el,col){

  ctx.font='190px "Archivo Black"'; ctx.textAlign='center'; ctx.fillStyle=flash?'#7CFC9A':col; ctx.fillText(txt,cx,base);

}

function bubbleK56(b,tt){
  const k=(tt-b[0])/FLY56; if(k<0||k>=1) return;
  const [x0,y0,col]=SRC56[b[2]], p=ease(k);
  const x=lerp(x0,320,p), y=lerp(y0,650,p)-Math.sin(p*Math.PI)*170, sc=lerp(0.55,1,Math.min(1,k*3))*lerp(1,0.8,p);
  const isCat=(b[2]==='cat'), size=(b[1].length>4)?38:58;
  ctx.save(); ctx.translate(x,y); ctx.scale(sc,sc); ctx.lineJoin='round';
  ctx.font=size+'px "Archivo Black"'; const tw=ctx.measureText(b[1]).width, w=tw+56+(isCat?58:0), h=88;
  ctx.fillStyle=col; ctx.strokeStyle=INK; ctx.lineWidth=7; rr(-w/2,-h/2,w,h,32); ctx.fill(); ctx.stroke();
  ctx.beginPath(); ctx.moveTo(-w/2+30,h/2-2); ctx.lineTo(-w/2+14,h/2+24); ctx.lineTo(-w/2+58,h/2-2); ctx.closePath(); ctx.fill(); ctx.stroke();
  ctx.fillStyle=col; ctx.fillRect(-w/2+34,h/2-8,22,12);

  if(isCat){ const cx=-w/2+44; ctx.fillStyle='#0b0613'; ctx.beginPath(); ctx.arc(cx,4,22,0,7); ctx.fill();
    ctx.beginPath(); ctx.moveTo(cx-20,-6); ctx.lineTo(cx-14,-30); ctx.lineTo(cx-2,-14); ctx.closePath(); ctx.fill();

  ctx.fillStyle=INK; ctx.textAlign='center'; ctx.textBaseline='middle'; ctx.fillText(b[1],tx,4); ctx.textBaseline='alphabetic';
  ctx.restore();
}
